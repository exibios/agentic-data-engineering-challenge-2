SELECT
  job_id,
  user_email,
  creation_time,
  start_time,
  end_time,
  state,
  total_bytes_billed,
  total_slot_ms,
  query
FROM
  `region-US`.INFORMATION_SCHEMA.JOBS_BY_USER -- Change region if necessary (e.g., region-EU)
WHERE
  creation_time >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 7 DAY)
  AND (
    UPPER(query) LIKE '%ML.GENERATE_TEXT%'
    OR UPPER(query) LIKE '%VECTOR_SEARCH%'
  )
ORDER BY
  creation_time DESC;
